let profile=JSON.parse(localStorage.getItem("cf_profile")||"null");
let done=JSON.parse(localStorage.getItem("cf_done")||"[]");
let data=null,lessonIndex=0,tab="text";

function clean(x){return String(x||"").toLowerCase().trim()}
function esc(x){return String(x).replace(/[&<>"']/g,c=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c]))}
function go(id){document.getElementById(id).scrollIntoView({behavior:"smooth"})}
function save(){localStorage.setItem("cf_profile",JSON.stringify(profile));localStorage.setItem("cf_done",JSON.stringify(done))}
function getCurriculum(c){
 let x=clean(c);
 if(x.includes("ai")||x.includes("machine learning")||x.includes("ml"))return CURRICULUM["ai engineer"];
 if(x.includes("data"))return CURRICULUM["data scientist"];
 if(x.includes("cyber")||x.includes("security"))return CURRICULUM["cyber security"];
 if(x.includes("devops")||x.includes("cloud"))return CURRICULUM["cloud devops"];
 if(x.includes("ui")||x.includes("ux")||x.includes("design"))return CURRICULUM["ui ux designer"];
 if(x.includes("web")||x.includes("frontend")||x.includes("full stack"))return CURRICULUM["web developer"];
 if(x.includes("software")||x.includes("programming")||x.includes("developer"))return CURRICULUM["software engineer"];
 return GENERIC;
}
function createProfile(){
 profile={
  education:document.getElementById("education").value||"Not specified",
  career:document.getElementById("career").value||"Software Developer",
  level:document.getElementById("level").value,
  hours:+document.getElementById("hours").value,
  goal:document.getElementById("goal").value,
  language:document.getElementById("language").value,
  skills:document.getElementById("skills").value.split(",").map(x=>x.trim()).filter(Boolean),
  need:document.getElementById("need").value||"General career guidance"
 };
 done=[];save();renderAll();
}
function renderAll(){
 data=getCurriculum(profile.career);
 document.getElementById("formArea").classList.add("hidden");
 document.getElementById("profileBox").classList.remove("hidden");
 document.getElementById("profileBox").innerHTML=`<div class="notice">🤖 <b>${esc(profile.career)}</b> path for ${esc(profile.education)} • ${esc(profile.level)} • ${profile.hours}h/day • ${esc(profile.goal)}<br>${esc(profile.need)}</div>`;
 ["dashboard","roadmap","learn","today","chat"].forEach(x=>document.getElementById(x).classList.remove("hidden"));
 document.getElementById("dCareer").textContent=profile.career;
 document.getElementById("dGoal").textContent=profile.goal+" target";
 document.getElementById("dHours").textContent=profile.hours+"h";
 let known=profile.skills.map(clean);
 let missing=data.skills.filter(s=>!known.some(k=>k.includes(clean(s))||clean(s).includes(k)));
 document.getElementById("gap").innerHTML=(missing.length?missing:["Focus on projects and proof-of-work"]).map(x=>`<span class="chip">+ ${esc(x)}</span>`).join("");
 document.getElementById("meta").textContent=data.lessons.length+" learning stages";
 renderRoadmap();renderLessonList();renderLesson();renderToday();updateProgress();initChat();
}
function renderRoadmap(){
 document.getElementById("roadmapList").innerHTML=data.lessons.map((l,i)=>`
 <div class="stage ${done.includes(i)?"done":""}">
  <input type="checkbox" ${done.includes(i)?"checked":""} onchange="toggleDone(${i})">
  <div><strong>Stage ${i+1}: ${esc(l.title)}</strong><div class="small">Learn → watch → take notes → practice</div></div>
  <div class="actions"><button class="btn" onclick="openLesson(${i})">Study inside site</button></div>
 </div>`).join("");
}
function toggleDone(i){done.includes(i)?done=done.filter(x=>x!==i):done.push(i);save();renderRoadmap();updateProgress();renderToday()}
function updateProgress(){
 let p=Math.round(done.length/data.lessons.length*100);
 document.getElementById("dProgress").textContent=p+"%";
 document.getElementById("bar").style.width=p+"%";
}
function renderLessonList(){
 document.getElementById("lessonList").innerHTML=data.lessons.map((l,i)=>`
 <button class="lesson-item ${i===lessonIndex?"active":""}" onclick="openLesson(${i})">
  <b>${i+1}. ${esc(l.title)}</b><div class="small">${done.includes(i)?"✓ Completed":"Not completed"}</div>
 </button>`).join("");
}
function openLesson(i){lessonIndex=i;tab="text";renderLessonList();renderLesson();go("learn")}
function renderLesson(){
 let l=data.lessons[lessonIndex];if(!l)return;
 document.getElementById("lessonTitle").textContent=l.title;
 document.getElementById("lessonSub").textContent=`Stage ${lessonIndex+1} of ${data.lessons.length}`;
 let body=document.getElementById("viewerBody");
 if(tab==="text"){
  body.innerHTML=`<article class="lesson-text"><h2>${esc(l.title)}</h2>${l.text.map((p,i)=>`<p><b>${i===0?"Concept":"Step "+(i+1)+":"}</b> ${esc(p)}</p>`).join("")}<div class="note">💡 <b>Learning rule:</b> Don't just watch. Pause, reproduce the idea yourself, then complete the practice task.</div></article>`;
 }else if(tab==="video"){
  body.innerHTML=`<div class="video-wrap"><iframe src="${l.video}?rel=0" title="${esc(l.title)}" allow="accelerometer; autoplay; clipboard-write; encrypted-media; gyroscope; picture-in-picture; web-share" allowfullscreen></iframe></div><p class="small">Video is embedded from YouTube. You stay inside CareerForge while watching.</p>`;
 }else if(tab==="notes"){
  body.innerHTML=`<article class="lesson-text"><h2>📝 Notes — ${esc(l.title)}</h2><pre>${esc(l.notes)}</pre><button class="primary" onclick="downloadNotes()">Save Notes</button></article>`;
 }else if(tab==="practice"){
  body.innerHTML=`<article class="lesson-text"><h2>🧪 Practice Lab</h2><p>Complete these without copying a finished answer.</p><ol>${l.practice.map(x=>`<li>${esc(x)}</li>`).join("")}</ol><div class="note">When finished, mark Stage ${lessonIndex+1} complete in the roadmap.</div></article>`;
 }else{
  body.innerHTML=`<article class="lesson-text"><h2>🔗 Embedded Source</h2><p>The video above is embedded from YouTube. The written lesson and notes are maintained by CareerForge.</p><div class="notice">Some third-party sites intentionally prevent embedding with browser security headers. CareerForge does not bypass those protections. Instead, it keeps its own learning layer here and embeds only content providers that allow it.</div></article>`;
 }
 document.querySelectorAll(".tabs button").forEach((b,i)=>b.classList.toggle("active",["text","video","notes","practice","sources"][i]===tab));
}
function showTab(t,btn){tab=t;renderLesson()}
function downloadNotes(){
 let l=data.lessons[lessonIndex];
 let blob=new Blob([l.title+"\\n\\n"+l.notes+"\\n\\nPractice:\\n- "+l.practice.join("\\n- ")],{type:"text/plain"});
 let a=document.createElement("a");a.href=URL.createObjectURL(blob);a.download=l.title.replace(/[^a-z0-9]+/gi,"-").toLowerCase()+"-notes.txt";a.click();URL.revokeObjectURL(a.href);
}
function renderToday(){
 let i=Math.min(done.length,data.lessons.length-1),l=data.lessons[i];
 document.getElementById("todayBox").innerHTML=`<h3>Today's best next step: ${esc(l.title)}</h3>
 <p class="muted">Use your ${profile.hours}h session like this:</p>
 <div class="chips"><span class="chip">30–40% Read lesson</span><span class="chip">30–40% Watch / reproduce</span><span class="chip">20–30% Practice</span></div>
 <div class="actions" style="margin-top:12px"><button class="primary" onclick="openLesson(${i})">Start inside CareerForge</button></div>`;
}
function initChat(){
 let m=document.getElementById("messages");m.innerHTML="";
 bot(`Hi! I'm your CareerForge agent 🤖\\n\\nTarget: ${profile.career}\\nLevel: ${profile.level}\\nTime: ${profile.hours}h/day\\n\\nAsk me about your roadmap, today's lesson, projects, skills or study strategy.`);
}
function bot(x){let d=document.createElement("div");d.className="msg bot";d.textContent=x;document.getElementById("messages").appendChild(d);document.getElementById("messages").scrollTop=999999}
function user(x){let d=document.createElement("div");d.className="msg user";d.textContent=x;document.getElementById("messages").appendChild(d);document.getElementById("messages").scrollTop=999999}
function sendChat(){
 let inp=document.getElementById("chatInput"),q=inp.value.trim();if(!q)return;user(q);inp.value="";
 setTimeout(()=>bot(answer(q)),120);
}
function answer(q){
 let x=clean(q),next=data.lessons[Math.min(done.length,data.lessons.length-1)];
 if(x.includes("today")||x.includes("start"))return `Your next lesson is "${next.title}".\\n\\nOpen the Learning Studio and use Lesson → Video → Notes → Practice. That sequence is better than only watching a video.`;
 if(x.includes("project"))return `For "${next.title}", build a small project that proves the skill. Keep it small enough to finish in 1–3 sessions, then document what you learned.`;
 if(x.includes("roadmap"))return `Your path has ${data.lessons.length} stages and ${done.length} are complete. The next stage is "${next.title}".`;
 if(x.includes("video")){openLesson(Math.min(done.length,data.lessons.length-1));setTimeout(()=>showTab("video"),50);return "I opened your next lesson's embedded video inside CareerForge."}
 if(x.includes("note")){openLesson(Math.min(done.length,data.lessons.length-1));setTimeout(()=>showTab("notes"),50);return "I opened the notes for your next lesson inside CareerForge."}
 if(x.includes("practice")){openLesson(Math.min(done.length,data.lessons.length-1));setTimeout(()=>showTab("practice"),50);return "I opened the practice section for your next lesson."}
 if(x.includes("job")||x.includes("intern"))return `For a ${profile.goal} target, focus on proof-of-work: finish projects, publish them, explain your decisions, and practice role-specific questions. Don't collect resources without completing them.`;
 return `For your ${profile.career} goal, I'd prioritize "${next.title}".\\n\\nTell me something specific like:\\n• "Open my next video"\\n• "Give me a project"\\n• "What should I practice?"\\n• "Explain this topic simply"`;
}
document.getElementById("chatInput").addEventListener("keydown",e=>{if(e.key==="Enter")sendChat()});
function resetAll(){localStorage.removeItem("cf_profile");localStorage.removeItem("cf_done");location.reload()}

 renderAll();
}
